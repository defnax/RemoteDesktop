cmake_minimum_required(VERSION 3.20)
project(RemoteDesktop VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DESKTOP_QT_MAJOR "5" CACHE STRING "Qt major version matching the RetroShare host (5 or 6)")
set_property(CACHE DESKTOP_QT_MAJOR PROPERTY STRINGS 5 6)
if(DESKTOP_QT_MAJOR STREQUAL "6")
  find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets Network)
elseif(DESKTOP_QT_MAJOR STREQUAL "5")
  find_package(Qt5 5.12 REQUIRED COMPONENTS Core Gui Widgets Network)
else()
  message(FATAL_ERROR "DESKTOP_QT_MAJOR must be 5 or 6")
endif()
set(RETROSHARE_SOURCE_DIR "" CACHE PATH "RetroShare source tree used to build the host")
set(RETROSHARE_CORE_LIBRARY "" CACHE FILEPATH "Windows RetroShare core import library")
set(RETROSHARE_GUI_LIBRARY "" CACHE FILEPATH "Windows RetroShare GUI import library")
set(RETROSHARE_COMPILE_DEFINITIONS "" CACHE STRING "ABI feature definitions from host build")
option(BUILD_RETROSHARE_PLUGIN "Build RetroShare module" ON)
option(DESKTOP_USE_BUNDLED_LIBVNC "Build the optional third_party/libvncserver checkout instead of installed packages" OFF)
if(DESKTOP_USE_BUNDLED_LIBVNC)
  set(WITH_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(WITH_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(WITH_FFMPEG OFF CACHE BOOL "" FORCE)
  set(WITH_SASL OFF CACHE BOOL "" FORCE)
  set(WITH_GNUTLS OFF CACHE BOOL "" FORCE)
  set(WITH_GCRYPT OFF CACHE BOOL "" FORCE)
  set(WITH_TIGHTVNC_FILETRANSFER OFF CACHE BOOL "" FORCE)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/libvncserver/CMakeLists.txt")
    message(FATAL_ERROR "Bundled mode requires third_party/libvncserver. Fetch it with scripts/fetch-libvnc.ps1 or use the default installed-package mode; see README.md")
  endif()
  add_subdirectory(third_party/libvncserver EXCLUDE_FROM_ALL)
  set(DESKTOP_VNC_LIBRARIES vncclient vncserver)
else()
  find_package(LibVNCServer 0.9.13 CONFIG QUIET)
  if(TARGET LibVNCServer::vncclient AND TARGET LibVNCServer::vncserver)
    set(DESKTOP_VNC_LIBRARIES LibVNCServer::vncclient LibVNCServer::vncserver)
  else()
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
      pkg_check_modules(DESKTOP_VNC QUIET IMPORTED_TARGET libvncclient>=0.9.13 libvncserver>=0.9.13)
    endif()
    if(NOT TARGET PkgConfig::DESKTOP_VNC)
      message(FATAL_ERROR "Install LibVNC development packages (MSYS2: mingw-w64-x86_64-libvncserver; Ubuntu: libvncserver-dev; Arch: libvncserver), set CMAKE_PREFIX_PATH, or enable DESKTOP_USE_BUNDLED_LIBVNC=ON. See README.md.")
    endif()
    set(DESKTOP_VNC_LIBRARIES PkgConfig::DESKTOP_VNC)
  endif()
endif()
# Sharing side: loopback VNC server and native input (used by the plugin).
add_library(desktop_engine STATIC src/VncWorker.cpp src/NativeInput.cpp)
target_include_directories(desktop_engine PUBLIC src)
# Viewing side: LibVNC client used by the plugin's viewer window.
add_library(desktop_client STATIC viewer/VncClient.cpp)
target_include_directories(desktop_client PUBLIC src viewer)
foreach(lib desktop_engine desktop_client)
  if(DESKTOP_USE_BUNDLED_LIBVNC)
    target_include_directories(${lib} PUBLIC third_party/libvncserver/include "${CMAKE_CURRENT_BINARY_DIR}/third_party/libvncserver/include")
  endif()
  target_link_libraries(${lib} PUBLIC Qt${DESKTOP_QT_MAJOR}::Widgets Qt${DESKTOP_QT_MAJOR}::Network ${DESKTOP_VNC_LIBRARIES})
  if(WIN32)
    target_link_libraries(${lib} PUBLIC user32 ws2_32)
  endif()
endforeach()
if(APPLE)
  target_link_libraries(desktop_engine PUBLIC "-framework ApplicationServices")
elseif(NOT WIN32)
  find_package(X11 REQUIRED)
  if(NOT X11_XTest_FOUND)
    message(FATAL_ERROR "Install libxtst development headers for X11 input")
  endif()
  target_include_directories(desktop_engine PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(desktop_engine PUBLIC ${X11_LIBRARIES} ${X11_XTest_LIB} ${CMAKE_DL_LIBS})
endif()
if(BUILD_RETROSHARE_PLUGIN)
  if(NOT EXISTS "${RETROSHARE_SOURCE_DIR}/libretroshare/src/retroshare/rsplugin.h")
    message(FATAL_ERROR "Set RETROSHARE_SOURCE_DIR to your host source tree")
  endif()
  add_library(RemoteDesktop MODULE src/Plugin.cpp src/Service.cpp src/SessionPage.cpp src/HostDialogs.cpp src/ChatWindow.cpp
              src/DesktopView.cpp viewer/ViewerWindow.cpp viewer/SystemKeys.cpp src/images.qrc)
  target_include_directories(RemoteDesktop PRIVATE "${RETROSHARE_SOURCE_DIR}/libretroshare/src" "${RETROSHARE_SOURCE_DIR}/retroshare-gui/src" "${RETROSHARE_SOURCE_DIR}/supportlibs/rapidjson/include")
  target_compile_definitions(RemoteDesktop PRIVATE ${RETROSHARE_COMPILE_DEFINITIONS})
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(RemoteDesktop PRIVATE -Wno-deprecated-declarations -Wno-cpp)
    if(CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 15)
      target_compile_options(RemoteDesktop PRIVATE -Wno-error=template-body)
    endif()
  endif()
  target_link_libraries(RemoteDesktop PRIVATE desktop_engine desktop_client)
  if(WIN32)
    target_compile_definitions(RemoteDesktop PRIVATE WINDOWS_SYS WIN32_LEAN_AND_MEAN)
    if(NOT RETROSHARE_CORE_LIBRARY OR NOT RETROSHARE_GUI_LIBRARY)
      message(FATAL_ERROR "Set both RetroShare import library paths on Windows")
    endif()
    target_link_libraries(RemoteDesktop PRIVATE "${RETROSHARE_CORE_LIBRARY}" "${RETROSHARE_GUI_LIBRARY}")
  elseif(APPLE)
    target_link_options(RemoteDesktop PRIVATE "-undefined" "dynamic_lookup")
  endif()
endif()
